function [iA_total,iB_total,iC_total,iA_R_true,iB_R_true,iC_R_true] = dynamic_current(uA,uB, uC, iA_R_raw, iB_R_raw,iC_R_raw,Cs1_pF,Cs2_pF,B_fault_scale,noiseA,noiseB,noiseC, Ts_model, Cself_pF)
%#codegen
persistent a1 b1 c1 a2 b2 c2 count
if isempty(count), a1=uA; b1=uB; c1=uC; a2=uA; b2=uB; c2=uC; count=uint8(0); end
if count==0
 duA=0; duB=0; duC=0; count=uint8(1);
elseif count==1
 duA=(uA-a1)/Ts_model; duB=(uB-b1)/Ts_model; duC=(uC-c1)/Ts_model; count=uint8(2);
else
 duA=(3*uA-4*a1+a2)/(2*Ts_model); duB=(3*uB-4*b1+b2)/(2*Ts_model); duC=(3*uC-4*c1+c2)/(2*Ts_model);
end
a2=a1; b2=b1; c2=c1; a1=uA; b1=uB; c1=uC;
iA_R_true=-iA_R_raw; iB_R_true=-B_fault_scale*iB_R_raw; iC_R_true=-iC_R_raw;
Cs1=Cs1_pF*1e-12; Cs2=Cs2_pF*1e-12; Cself=Cself_pF*1e-12;
iA_total=Cself*duA+Cs1*(duA-duB)+iA_R_true+noiseA;
iB_total=Cself*duB+Cs1*(duB-duA)+Cs2*(duB-duC)+iB_R_true+noiseB;
iC_total=Cself*duC+Cs2*(duC-duB)+iC_R_true+noiseC;
end

function i = moa_current(v, Vref_moa, Iref_moa, alpha_moa)
% 氧化锌避雷器非线性阻性电流模型（参数来自统一 cfg）
i = Iref_moa * sign(v) * (abs(v)/Vref_moa)^alpha_moa;
end
